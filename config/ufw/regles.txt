# Règles du pare-feu UFW, dans l'ordre où je les ai appliquées
# (le port 22 est autorisé AVANT "ufw enable" pour ne pas couper ma propre connexion SSH)

sudo ufw default deny incoming
sudo ufw default allow outgoing
sudo ufw allow 22/tcp
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw enable

# Résultat de "sudo ufw status verbose" :
# Status: active
# Default: deny (incoming), allow (outgoing), disabled (routed)
# 22/tcp, 80/tcp, 443/tcp      ALLOW IN    Anywhere (IPv4 et IPv6)
